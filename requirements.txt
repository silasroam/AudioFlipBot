# --- Pyrogram-клиент (MTProto вместо Bot API: файлы до 2000 МБ) ---
# Взят maintained-форк Pyrogram: тот же API и тот же импорт `pyrogram`,
# но он работает на Python 3.13/3.14 (официальный pyrogram 2.0.106 на 3.13+
# даже не импортируется: pyrogram/sync.py дергает asyncio.get_event_loop(),
# который удалён в Python 3.14).
kurigram>=2.2.26

# Если у тебя Python 3.10-3.12, можно вместо kurigram поставить официальный:
# pyrogram==2.0.106

# Ускоритель AES (необязателен, но с ним скачивание/загрузка заметно быстрее).
tgcrypto>=1.2.5

# --- Мини-HTTP-сервер для Render Web Service (Free) ---
# Render требует, чтобы у Web Service был открыт порт, иначе сервис падает с
# "No open ports detected". Бот поднимает крошечный aiohttp-фреймворк на $PORT
# (по умолчанию 10000) с маршрутами / и /health, которые отвечают 200 OK.
# На работу бота это не влияет: конвертация, PENDING и ffmpeg не трогаются.
aiohttp>=3.9

# --- SQLite для учёта обработанных файлов (нумерация имён) ---
# Таблица users_files(user_id, file_count) создаётся автоматически при старте.
# Асинхронный драйвер, чтобы работа с БД не блокировала event loop бота.
aiosqlite>=0.19
